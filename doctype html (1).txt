function getAparStatus(expDateStr) {
  if (!expDateStr || expDateStr.trim() === "" || expDateStr.toUpperCase() === "TIDAK ADA") {
    return { code: "EMPTY", label: "Kosong / N/A", badge: "bg-slate-100 text-slate-600" };
  }
  const parsed = new Date(expDateStr);
  if (isNaN(parsed.getTime())) {
    return { code: "EMPTY", label: expDateStr, badge: "bg-slate-100 text-slate-600" };
  }
  const today = new Date();
  const diffDays = Math.ceil((parsed - today) / (1000 * 60 * 60 * 24));
  if (diffDays < 0) {
    return { code: "EXPIRED", label: `${expDateStr} (Expired)`, badge: "bg-rose-100 text-rose-700 font-semibold" };
  } else if (diffDays <= 90) {
    return { code: "SOON", label: `${expDateStr} (${diffDays}h)`, badge: "bg-amber-100 text-amber-800 font-semibold" };
  } else {
    return { code: "VALID", label: expDateStr, badge: "bg-emerald-100 text-emerald-700" };
  }
}

function toNum(val) {
  if (!val) return 0;
  const cleaned = String(val).replace(/,/g, ".").replace(/[^0-9.-]/g, "");
  const n = parseFloat(cleaned);
  return isNaN(n) ? 0 : n;
}

function populateFilterDropdowns() {
  const fillSelect = (id, key, defaultLabel) => {
    const sel = document.getElementById(id);
    const currentVal = sel.value;
    const uniqueVals = [...new Set(storeData.map(d => (d[key] || "").trim()).filter(Boolean))].sort();
    sel.innerHTML = `<option value="">${defaultLabel}</option>` +
      uniqueVals.map(v => `<option value="${v}">${v}</option>`).join("");
    sel.value = uniqueVals.includes(currentVal) ? currentVal : "";
  };

  fillSelect("filterBU", "BUSINESS UNIT", "Semua Business Unit");
  fillSelect("filterBrand", "BRAND", "Semua Brand");
  fillSelect("filterKota", "KABUPATEN / KOTA", "Semua Kabupaten/Kota");
  fillSelect("filterSubCat", "SUB CATEGORY", "Semua Sub Category");
}

function applyFilters() {
  const q = document.getElementById("filterSearch").value.toLowerCase().trim();
  const bu = document.getElementById("filterBU").value;
  const brand = document.getElementById("filterBrand").value;
  const kota = document.getElementById("filterKota").value;
  const subCat = document.getElementById("filterSubCat").value;
  const aparStat = document.getElementById("filterAparStatus").value;

  filteredData = storeData.filter(row => {
    if (bu && row["BUSINESS UNIT"] !== bu) return false;
    if (brand && row["BRAND"] !== brand) return false;
    if (kota && row["KABUPATEN / KOTA"] !== kota) return false;
    if (subCat && row["SUB CATEGORY"] !== subCat) return false;
    if (aparStat) {
      const st = getAparStatus(row["EXPIRED APAR"]).code;
      if (st !== aparStat) return false;
    }
    if (q) {
      const hay = [
        row["KODE STORE"], row["BRAND+LOCATION NAME"], row["SITE NAME"],
        row["BRAND"], row["STORE MANAGER"], row["LP SPV"], row["LP ARH"], row["KETERANGAN"]
      ].join(" ").toLowerCase();
      if (!hay.includes(q)) return false;
    }
    return true;
  });

  currentPage = 1;
  updateKPIs();
  updateCharts();
  renderTable();
}

  document.getElementById("filterSearch").value = "";

  document.getElementById("filterBrand").value = "";
  document.getElementById("filterKota").value = "";
  document.getElementById("filterSubCat").value = "";
  document.getElementById("filterAparStatus").value = "";
  applyFilters();
}

function updateKPIs() {
  const total = filteredData.length;
  const jaksel = filteredData.filter(d => (d["KABUPATEN / KOTA"] || "").includes("SELATAN")).length;
  const jaktim = filteredData.filter(d => (d["KABUPATEN / KOTA"] || "").includes("TIMUR")).length;

  const totalArea = filteredData.reduce((acc, d) => acc + toNum(d["AREA (m²)"]), 0);
  const avgArea = total ? Math.round(totalArea / total) : 0;

  const totalMpp = filteredData.reduce((acc, d) => acc + toNum(d["JUMLAH MPP"]), 0);
  const storeWithMpp = filteredData.filter(d => toNum(d["JUMLAH MPP"]) > 0).length;

  const totalCam = filteredData.reduce((acc, d) => acc + toNum(d["CAMERA"]), 0);
  const totalDvr = filteredData.reduce((acc, d) => acc + toNum(d["DVR HARDDISK"]), 0);

  const totalApar = filteredData.reduce((acc, d) => acc + toNum(d["APAR"]), 0);
  const aparTerpasang = filteredData.reduce((acc, d) => acc + toNum(d["APAR TERPASANG"]), 0);
  const aparBackup = filteredData.reduce((acc, d) => acc + toNum(d["APAR BACKUP"]), 0);

  let expCount = 0, soonCount = 0;
  filteredData.forEach(d => {
    const st = getAparStatus(d["EXPIRED APAR"]).code;
    if (st === "EXPIRED") expCount++;
    if (st === "SOON") soonCount++;
  });

  document.getElementById("kpiTotalStore").textContent = total;
  document.getElementById("kpiAreaSub").textContent = `Jaksel: ${jaksel} | Jaktim: ${jaktim}`;
  document.getElementById("kpiTotalArea").textContent = `${totalArea.toLocaleString("id-ID")} m²`;
  document.getElementById("kpiAvgArea").textContent = `Rata-rata: ${avgArea.toLocaleString("id-ID")} m²`;
  document.getElementById("kpiTotalMpp").textContent = totalMpp;
  document.getElementById("kpiStoreWithMpp").textContent = `Store dengan MPP: ${storeWithMpp}`;
  document.getElementById("kpiTotalCam").textContent = totalCam.toLocaleString("id-ID");
  document.getElementById("kpiTotalDvr").textContent = `Total DVR/NVR: ${totalDvr}`;
  document.getElementById("kpiTotalApar").textContent = totalApar;
  document.getElementById("kpiAparSub").textContent = `Terpasang: ${aparTerpasang} | Backup: ${aparBackup}`;
  document.getElementById("kpiAparExpired").textContent = expCount;
  document.getElementById("kpiAparSoon").textContent = `Segera Expired (<90 hr): ${soonCount}`;
}

    regCounts[lbl] = (regCounts[lbl] || 0) + 1;
  });

  renderChart("chartRegion", "doughnut", {
    labels: Object.keys(regCounts),
    datasets: [{
      data: Object.values(regCounts),
      backgroundColor: palette
    }]
  }, { plugins: { legend: { position: "bottom", labels: { boxWidth: 10, font: { size: 10 } } } } });

  // 3. Business Unit Chart
  const buCounts = {};
  filteredData.forEach(d => {
    const bu = (d["BUSINESS UNIT"] || "N/A").replace("PT. ", "").replace(", Tbk", "");
    buCounts[bu] = (buCounts[bu] || 0) + 1;
  });

  renderChart("chartBU", "bar", {
    labels: Object.keys(buCounts),
    datasets: [{
      label: "Store",
      data: Object.values(buCounts),
      backgroundColor: "#4f46e5",
      borderRadius: 4
    }]
  }, { plugins: { legend: { display: false } }, scales: { x: { ticks: { font: { size: 9 } } } } });

  // 4. APAR Status Chart
  const aparGroups = { Valid: 0, "Segera Expired (<90h)": 0, Expired: 0, "Data Kosong": 0 };
  filteredData.forEach(d => {
    const c = getAparStatus(d["EXPIRED APAR"]).code;
    if (c === "VALID") aparGroups.Valid++;
    else if (c === "SOON") aparGroups["Segera Expired (<90h)"]++;
    else if (c === "EXPIRED") aparGroups.Expired++;
    else aparGroups["Data Kosong"]++;
  });

  renderChart("chartApar", "pie", {
    labels: Object.keys(aparGroups),
    datasets: [{
      data: Object.values(aparGroups),
      backgroundColor: ["#10b981", "#f59e0b", "#e11d48", "#94a3b8"]
    }]
  }, { plugins: { legend: { position: "bottom", labels: { boxWidth: 10, font: { size: 10 } } } } });
}

function renderChart(canvasId, type, data, extraOptions = {}) {
  const ctx = document.getElementById(canvasId).getContext("2d");
  if (charts[canvasId]) {
    charts[canvasId].destroy();
  }
  charts[canvasId] = new Chart(ctx, {
    type,
    data,
    options: {
      responsive: true,
      maintainAspectRatio: false,
      ...extraOptions
    }
  });
}

function openFormModal(mode, kodeStore = "") {
  const form = document.getElementById("storeForm");
  form.reset();
  switchFormTab(0);
  document.getElementById("formMode").value = mode;
  document.getElementById("originalKodeStore").value = kodeStore;

  if (mode === "edit") {
    document.getElementById("formModalTitle").textContent = `Edit Data Store: ${kodeStore}`;
    const row = storeData.find(d => d["KODE STORE"] === kodeStore);
    if (row) {
      ALL_COLUMNS.forEach(col => {
        const el = form.elements[col];
        if (el) el.value = row[col] || "";
      });
    }
  } else {
    document.getElementById("formModalTitle").textContent = "Tambah Data Store Baru";
    form.elements["STATUS"].value = "ACTIVE";
    form.elements["PROVINSI"].value = "DKI JAKARTA";
    form.elements["PULAU"].value = "JAWA";
  }

  showModal("formModal");
}

async function handleFormSubmit(e) {
  e.preventDefault();
  const form = e.target;
  const mode = document.getElementById("formMode").value;
  const origKode = document.getElementById("originalKodeStore").value;

  const newRecord = {};
  ALL_COLUMNS.forEach(col => {
    const el = form.elements[col];
    newRecord[col] = el ? el.value.trim() : "";
  });

  if (!newRecord["KODE STORE"]) {
    showToast("KODE STORE wajib diisi!", "error");
    return;
  }

  if (mode === "create") {
    if (storeData.some(d => d["KODE STORE"] === newRecord["KODE STORE"])) {
      showToast("KODE STORE sudah terdaftar!", "error");
      return;
    }
    storeData.unshift(newRecord);
    saveToLocalStorage();
    populateFilterDropdowns();
    applyFilters();
    closeModal("formModal");
    showToast(`Store ${newRecord["KODE STORE"]} berhasil ditambahkan.`, "success");
    await syncCrudToGas("create", { record: newRecord });
  } else {
    const idx = storeData.findIndex(d => d["KODE STORE"] === origKode);
    if (idx !== -1) {
      storeData[idx] = newRecord;
      saveToLocalStorage();
      populateFilterDropdowns();
      applyFilters();
      closeModal("formModal");
      showToast(`Store ${newRecord["KODE STORE"]} berhasil diperbarui.`, "success");
      await syncCrudToGas("update", { originalKode: origKode, record: newRecord });
    }
  }
}

async function deleteStore(kodeStore) {
  if (!confirm(`Yakin ingin menghapus data store "${kodeStore}"? Tindakan ini juga akan menghapus baris terkait di Google Sheet jika sinkronisasi aktif.`)) {
    return;
  }
  storeData = storeData.filter(d => d["KODE STORE"] !== kodeStore);
  saveToLocalStorage();
  populateFilterDropdowns();
  applyFilters();
  showToast(`Store ${kodeStore} telah dihapus.`, "success");
  await syncCrudToGas("delete", { kodeStore });
}

function handleCsvUpload(event) {
  const file = event.target.files[0];
  if (!file) return;

  Papa.parse(file, {
    header: true,
    skipEmptyLines: true,
    complete: function(results) {
      const imported = results.data.map(raw => {
        const cleanRow = {};
        ALL_COLUMNS.forEach(col => {
          cleanRow[col] = raw[col] !== undefined && raw[col] !== null ? String(raw[col]).trim() : "";
        });
        return cleanRow;
      }).filter(r => r["KODE STORE"]);

      if (imported.length === 0) {
        showToast("File CSV kosong atau header tidak sesuai.", "error");
        return;
      }

      storeData = imported;
      saveToLocalStorage();
      populateFilterDropdowns();
      applyFilters();
      showToast(`Berhasil mengimpor ${imported.length} data store dari CSV!`, "success");
    }
  });
  event.target.value = "";
}

function exportToCsv() {
  const csv = Papa.unparse(filteredData, { columns: ALL_COLUMNS });
  const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `Data_Store_Jaksel_Jaktim_${new Date().toISOString().slice(0, 10)}.csv`;
  a.click();
  URL.revokeObjectURL(url);
}

  const dot = document.getElementById("syncDot");
  const text = document.getElementById("syncText");
  if (url && url.startsWith("http")) {
    dot.className = "w-2 h-2 rounded-full bg-emerald-400 animate-pulse";
    text.textContent = "Terhubung Google Sheet (Online)";
  } else {

    text.textContent = "Mode Lokal (Belum Terhubung GAS)";

async function pullFromGoogleSheet(silent = false) {
  const url = getGasUrl();
  if (!url || !url.startsWith("http")) {
    if (!silent) openGasModal();
    return;
  }
  if (!silent) showToast("Menarik data terbaru dari Google Spreadsheet...", "info");
  try {
    const res = await fetch(url);
    const json = await res.json();
    if (json.status === "success" && Array.isArray(json.data)) {
      if (json.data.length > 0) {
        storeData = json.data.map(raw => {
          const cleanRow = {};
          ALL_COLUMNS.forEach(col => {
            cleanRow[col] = raw[col] !== undefined && raw[col] !== null ? String(raw[col]).trim() : "";
          });
          return cleanRow;
        });
        saveToLocalStorage();
        populateFilterDropdowns();
        applyFilters();
      }
      if (!silent) showToast(`Sinkronisasi Pull berhasil (${storeData.length} store dimuat).`, "success");
    } else {
      throw new Error(json.message || "Format respon GAS tidak valid.");
    }
  } catch (err) {
    if (!silent) showToast("Gagal Pull dari Google Sheet: " + err.message, "error");
  }
}

async function pushAllToGoogleSheet() {
  const url = getGasUrl();
  if (!url || !url.startsWith("http")) {
    openGasModal();
    return;
  }
  if (!confirm(`Kirim seluruh ${storeData.length} baris data lokal ini untuk menimpa isi Google Spreadsheet?`)) return;

  showToast("Mengirim seluruh data ke Google Spreadsheet...", "info");
  try {
    const res = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify({ action: "syncAll", records: storeData })
    });
    const json = await res.json();
    if (json.status === "success") {
      showToast(json.message, "success");
    } else {
      throw new Error(json.message);
    }
  } catch (err) {
    showToast("Gagal Push All: " + err.message, "error");
  }
}

function showModal(id) {
  const el = document.getElementById(id);
  el.classList.remove("hidden");
  el.classList.add("flex");
}
function closeModal(id) {
  const el = document.getElementById(id);
  el.classList.add("hidden");
  el.classList.remove("flex");
}
function showToast(msg, type = "info") {
  const toast = document.getElementById("toast");
  const icon = document.getElementById("toastIcon");
  const text = document.getElementById("toastMessage");

  const colors = {
    success: "bg-emerald-900 border-emerald-600",
    error: "bg-rose-900 border-rose-600",
    info: "bg-slate-900 border-sky-600"
  };
  toast.className = `fixed bottom-5 right-5 z-50 flex items-center gap-2.5 px-4 py-3 rounded-xl shadow-xl text-xs font-medium text-white border ${colors[type] || colors.info}`;
  icon.textContent = type === "success" ? "✔" : type === "error" ? "✖" : "ℹ";
  text.textContent = msg;

  toast.classList.remove("hidden");
  clearTimeout(window._toastTimer);
  window._toastTimer = setTimeout(() => toast.classList.add("hidden"), 4200);
}
